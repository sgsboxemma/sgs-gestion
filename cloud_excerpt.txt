    try {

      for (const [path, item] of Object.entries(obj || {})) {

    } catch (_) {}

    } catch (_) {}

        }
      }
    } catch (_) {}
  }

  async function loadImageElement(file) {
    return new Promise((resolve, reject) => {
      const url = URL.createObjectURL(file);
      const img = new Image();
      img.onload = () => { URL.revokeObjectURL(url); resolve(img); };
      img.onerror = () => { URL.revokeObjectURL(url); reject(new Error("Image illisible.")); };
      img.src = url;
    });
  }

  async function canvasBlob(canvas, quality) {
    return new Promise(resolve => canvas.toBlob(resolve, "image/jpeg", quality));
  }

  async function optimizeImage(file, kind) {
    if (!file || !String(file.type || "").startsWith("image/")) return file;
    if (/image\/(gif|svg\+xml)/i.test(file.type || "")) return file;
    try {
      const img = await loadImageElement(file);
      const isPhoto = kind === "photo";
      const maxSide = isPhoto ? 900 : 1600;
      const targetBytes = isPhoto ? 150 * 1024 : 500 * 1024;
      const hardMaxBytes = isPhoto ? 280 * 1024 : 850 * 1024;
      const largest = Math.max(img.naturalWidth || img.width, img.naturalHeight || img.height);
      if (!largest) return file;

      let scale = Math.min(1, maxSide / largest);
      let width = Math.max(1, Math.round((img.naturalWidth || img.width) * scale));
      let height = Math.max(1, Math.round((img.naturalHeight || img.height) * scale));
      const canvas = document.createElement("canvas");
      const ctx = canvas.getContext("2d", { alpha: false });
      if (!ctx) return file;

      let best = null;
      for (let pass = 0; pass < 4; pass++) {
        canvas.width = width;
        canvas.height = height;
        ctx.fillStyle = "#ffffff";
        ctx.fillRect(0, 0, width, height);
        ctx.drawImage(img, 0, 0, width, height);

        const qualities = isPhoto ? [0.78, 0.70, 0.62, 0.54] : [0.84, 0.76, 0.68, 0.60];
        for (const quality of qualities) {
          const blob = await canvasBlob(canvas, quality);
          if (!blob) continue;
          if (!best || blob.size < best.size) best = blob;
          if (blob.size <= targetBytes) { best = blob; break; }
        }
        if (best && best.size <= targetBytes) break;
        width = Math.max(1, Math.round(width * 0.82));
        height = Math.max(1, Math.round(height * 0.82));
      }

      if (!best) return file;
      // Never increase the file size. For already-small images, keep the original.
      if (best.size >= file.size && file.size <= hardMaxBytes) return file;
      if (best.size > hardMaxBytes) throw new Error(isPhoto
        ? "La photo reste trop lourde après compression. Reprenez une photo moins détaillée."
        : "L’image du certificat reste trop lourde après compression.");
      const base = String(file.name || kind).replace(/\.[^.]+$/, "").replace(/[^a-zA-Z0-9._-]+/g, "-") || kind;
      return new File([best], `${base}.jpg`, { type: "image/jpeg", lastModified: file.lastModified || Date.now() });
    } catch (error) {
      if (error && /trop lourde après compression/.test(String(error.message || ""))) throw error;
      return file;
    }
  }

  async function cachedPhotoUrl(path) {
    if (!path) return "";
    const memoryUrl = photoObjectUrlMemory.get(path);
    if (memoryUrl) return memoryUrl;

    if ("caches" in window) {
      try {
        const cache = await caches.open(PHOTO_BLOB_CACHE);
        const key = photoCacheRequest(path);
        const cachedResponse = await cache.match(key);
        if (cachedResponse) {
          const cachedBlob = await cachedResponse.blob();
          const objectUrl = URL.createObjectURL(cachedBlob);
          photoObjectUrlMemory.set(path, objectUrl);
          return objectUrl;
        }

        // First display on this browser: download once, create a small local display copy,
        // and reuse it on subsequent sessions. The original file in Supabase is untouched.
        const signedUrl = await signed(path, 86400, true);
        if (!signedUrl) return "";
        const response = await fetch(signedUrl, { cache: "force-cache" });
        if (!response.ok) return "";
        const originalBlob = await response.blob();
        let displayBlob = originalBlob;
        try {
          const source = new File([originalBlob], "photo", { type: originalBlob.type || "image/jpeg" });
          const optimized = await optimizeImage(source, "photo");
          if (optimized && optimized.size <= originalBlob.size) displayBlob = optimized;
        } catch (_) {}
        await cache.put(key, new Response(displayBlob, {
          headers: {
            "Content-Type": displayBlob.type || "image/jpeg",
            "Cache-Control": "public, max-age=31536000, immutable"
          }
        }));
        const objectUrl = URL.createObjectURL(displayBlob);
        photoObjectUrlMemory.set(path, objectUrl);
        return objectUrl;
      } catch (_) {
        // Fall back to the signed URL if browser Cache Storage is unavailable or full.
      }
    }
    return signed(path, 86400, true);
  }

  async function upload(memberId, kind, file, limitMb) {
    if (!file) return null;
    if (kind === "photo" && !file.type.startsWith("image/")) throw new Error("La photo doit être une image.");
    if (kind === "medical" && !(file.type.startsWith("image/") || file.type === "application/pdf")) {
      throw new Error("Le certificat doit être une image ou un PDF.");
    }
    file = await optimizeImage(file, kind);
    // Hard post-optimization caps prevent a large unsupported image from entering Storage.
    if (kind === "photo" && file.size > 400 * 1024) {
      throw new Error("La photo reste trop lourde après optimisation (400 Ko maximum).");
    }
    if (kind === "medical" && String(file.type || "").startsWith("image/") && file.size > 1024 * 1024) {
      throw new Error("L’image du certificat reste trop lourde après optimisation (1 Mo maximum). Utilisez un PDF si nécessaire.");
    }
    if (file.size > limitMb * 1024 * 1024) throw new Error(`${file.name} dépasse ${limitMb} Mo après optimisation.`);
    const path = `${memberId}/${kind}-${crypto.randomUUID()}.${ext(file)}`;
    const { error } = await client.storage.from(BUCKET).upload(path, file, {
      upsert: false,
      contentType: file.type,
      // Unique filenames are used for every replacement, so a long cache is safe.
      cacheControl: "31536000"
    });
    if (error) throw error;
    return path;
  }

  async function removeFiles(paths, strict = false) {
    const clean = [...new Set(paths.filter(Boolean))];
    if (!clean.length) return;
    for (let i = 0; i < clean.length; i += 100) {
      const batch = clean.slice(i, i + 100);
      const { error } = await client.storage.from(BUCKET).remove(batch);
      if (!error) batch.forEach(invalidateSignedPath);
      if (error && strict) throw error;
    }
  }
